        - k_off_PS*C_EXG[1,j, Bound_ISM_mem]*V_ISM[j]
        - kint_PS*C_EXG[1,j, Bound_ISM_mem]*V_ISM[j]
         )/V_ISM[j] ;
                
      ;; Interstitial space
      d/dt(C_EXG[1..1,1..N_Organs,IntS]) =   
        ((1-sigma_V[j])*LF[j]*C_EXG[i,j,V] ;; going from vascular via Lymph 
        - CL_up[j]*(1-FR)*C_EXG[i,j,IntS] ;; pinocytosis 
        + CL_Up[j]*(1-FR)*C_EXG[i,j,ISM] ;; exocytosis
        -(1-Sigma_IS[j])*LF[j]*C_EXG[i,j,IntS];
        )/V_IntS[j] ;;
                        
      ;;;;;;;;;;;;;;;; For FcRn-IgG (bounded)
      d/dt(C_EXG[1,1..N_Organs,bound_VM]) = 
          (K_on_7_EXG*C_EXG[1,j,VM]*C_FcRn_VM[j]*V_VM[j] ;; rxn on 
        - K_off_7_EXG*C_EXG[1,j,Bound_VM]*V_VM[j] ;; rxn off
        - K_on_7_EXG2*C_EXG[1,j,Bound_VM]*C_FcRn_VM[j]*V_VM[j] ;; rxn on 
        + K_off_7_EXG2*C_EXG[1,j,Bound2_VM]*V_VM[j] ;; rxn off 
        + CL_UP[j]*FR*C_EXG[1,j,Bound_E7b]  ;; from Endosomal 6b to Vascular membrane
        - CL_UP[j]*FR*C_EXG[1,j,Bound_VM] ;; to Endosomal 7
        - kdeg_FcRn_Ab*C_EXG[1,j,Bound_VM]*V_VM[j] ;; recover FcRn when IgG-FrRn is destroyed
         )/V_VM[j];
                
      d/dt(C_EXG[1,1..N_Organs,Bound2_VM]) = 
          (K_on_7_EXG2*C_EXG[1,j,Bound_VM]*C_FcRn_VM[j]*V_VM[j] ;; rxn on 
        - K_off_7_EXG2*C_EXG[1,j,Bound2_VM]*V_VM[j] ;; rxn off 
        + CL_UP[j]*FR*C_EXG[1,j,Bound2_E7b]  ;; from Endosomal 6b to Vascular membrane
        - CL_UP[j]*FR*C_EXG[1,j,Bound2_VM] ;; to Endosomal 7
        - kdeg_FcRn_Ab*C_EXG[1,j,Bound2_VM]*V_VM[j] ;; recover FcRn when IgG-FrRn is destroyed
         )/V_VM[j] ;
                

      ;; endosomal space 
      d/dt(C_EXG[1,1..N_Organs,bound_E7]) = 
        (+ K_on_7_EXG*C_EXG[1,j,E7]*C_FcRn_E7[j]*V_E7[j] ;; rxn on 
        - K_off_7_EXG*C_EXG[1,j,bound_E7]*V_E7[j] ;; rxn off
        - K_on_7_EXG2*C_EXG[1,j,bound_E7]*C_FcRn_E7[j]*V_E7[j] ;; rxn on 
        + K_off_7_EXG2*C_EXG[1,j,bound2_E7]*V_E7[j] ;; rxn off 
        - CL_UP[j]*C_EXG[1,j,Bound_E7] ;; goes to E6
        + CL_UP[j]*FR*C_EXG[1,j,Bound_VM] ;; from VM 
        + CL_UP[j]*(1-FR)*C_EXG[1,j,Bound_ISM]  ;; from ISM
         )/V_E7[j] ;
                

      d/dt(C_EXG[1,1..N_Organs,bound2_E7]) = 
          (K_on_7_EXG2*C_EXG[1,j,bound_E7]*C_FcRn_E7[j]*V_E7[j] ;; rxn on 
        - K_off_7_EXG2*C_EXG[1,j,bound2_E7]*V_E7[j] ;; rxn off 
        - CL_UP[j]*C_EXG[1,j,Bound2_E7] ;; goes to E6
        + CL_UP[j]*FR*C_EXG[1,j,Bound2_VM] ;; from VM 
        + CL_UP[j]*(1-FR)*C_EXG[1,j,Bound2_ISM]   ;; from ISM
         )/V_E7[j] ;
                
      ;; endosomal pH=6 a
      d/dt(C_EXG[1,1..N_Organs,bound_E6a]) = 
        (+ K_on_6_EXG*C_EXG[1,j,E6a]*C_FcRn_E6a[j]*V_E6a[j] ;; rxn on 
        - K_off_6_EXG*C_EXG[1,j,bound_E6a]*V_E6a[j] ;; rxn off
        - K_on_6_EXG2*C_EXG[1,j,bound_E6a]*C_FcRn_E6a[j]*V_E6a[j] ;; rxn on 
        + K_off_6_EXG2*C_EXG[1,j,bound2_E6a]*V_E6a[j] ;; rxn off 
        + CL_UP[j]*C_EXG[1,j,Bound_E7] ;; internalization, from pH7 to pH6a
        - CL_up[j]*C_EXG[1,j,Bound_E6a] ;; exocytosis, from E6a to E7b
        )/V_E6a[j] ;
                
      d/dt(C_EXG[1,1..N_Organs,bound2_E6a]) = 
        (+ K_on_6_EXG2*C_EXG[1,j,bound_E6a]*C_FcRn_E6a[j]*V_E6a[j] ;; rxn on 
        - K_off_6_EXG2*C_EXG[1,j,bound2_E6a]*V_E6a[j] ;; rxn off 
        + CL_UP[j]*C_EXG[1,j,Bound2_E7] ;; internalization, from pH7 to pH6a
        - CL_up[j]*C_EXG[1,j,Bound2_E6a] ;; exocytosis, from E6a to E7b
        )/V_E6a[j];
                
      ;; endosomal pH=7.4  b
      d/dt(C_EXG[1,1..N_Organs, bound_E7b]) = 
        (+ K_on_7_EXG*C_EXG[1,j, E7b]*C_FcRn_E7b[j]*V_E7b[j] ;; rxn on 
        - K_off_7_EXG*C_EXG[1,j,bound_E7b]*V_E7b[j] ;; rxn off
        - K_on_7_EXG2*C_EXG[1,j, bound_E7b]*C_FcRn_E7b[j]*V_E7b[j] ;; rxn on 
        + K_off_7_EXG2*C_EXG[1,j,bound2_E7b]*V_E7b[j] ;; rxn off 
        + CL_UP[j]*C_EXG[1,j,Bound_E6a] ;; internalization, from E6a to E7b
        - CL_up[j]*C_EXG[1,j,Bound_E7b] ;; exocytosis, from E7b to vascular/interstitial membrane
         )/V_E7b[j];
                
      d/dt(C_EXG[1,1..N_Organs, bound2_E7b]) = 
        (+ K_on_7_EXG2*C_EXG[1,j, bound_E7b]*C_FcRn_E7b[j]*V_E7b[j] ;; rxn on 
        - K_off_7_EXG2*C_EXG[1,j,bound2_E7b]*V_E7b[j] ;; rxn off 
        + CL_UP[j]*C_EXG[1,j,Bound2_E6a] ;; internalization, from E6a to E7b
        - CL_up[j]*C_EXG[1,j,Bound2_E7b] ;; exocytosis, from E7b to vascular/interstitial membrane
         )/V_E7b[j];
                
      ;; Interstitial membrane
      d/dt(C_EXG[1,1..N_Organs,bound_ISM]) = 
          (K_on_7_EXG*C_EXG[1,j,ISM]*C_FcRn_ISM[j]*V_ISM[j] ;; rxn on  
        - K_off_7_EXG*C_EXG[1,j,Bound_ISM]*V_ISM[j] ;; rxn off
        - K_on_7_EXG2*C_EXG[1,j,Bound_ISM]*C_FcRn_ISM[j]*V_ISM[j] ;; rxn on 
        + K_off_7_EXG2*C_EXG[1,j,Bound2_ISM]*V_ISM[j] ;; rxn off 
        + CL_UP[j]*(1-FR)*C_EXG[1,j,Bound_E7b]  ;; from Endosomal 6 to interstitial membrane
        - CL_UP[j]*(1-FR)*C_EXG[1,j,Bound_ISM] ;; to Endosomal 7
        - kdeg_FcRn_Ab*C_EXG[1,j,Bound_ISM]*V_ISM[j]  ;; recover FcRn when IgG-FrRn is destroyed
         )/V_ISM[j];
                
      d/dt(C_EXG[1,1..N_Organs,bound2_ISM]) = 
        (+ K_on_7_EXG2*C_EXG[1,j,Bound_ISM]*C_FcRn_ISM[j]*V_ISM[j] ;; rxn on 
        - K_off_7_EXG2*C_EXG[1,j,Bound2_ISM]*V_ISM[j] ;; rxn off 
        + CL_UP[j]*(1-FR)*C_EXG[1,j,Bound2_E7b]  ;; from Endosomal 6 to interstitial membrane
        - CL_UP[j]*(1-FR)*C_EXG[1,j,Bound2_ISM] ;; to Endosomal 7
        - kdeg_FcRn_Ab*C_EXG[1,j,Bound2_ISM]*V_ISM[j]  ;; recover FcRn when IgG-FrRn is destroyed
         )/V_ISM[j];
                
  ;****************************************************************************************************      
        
      ;; Free Endogenous IgG 
      
        
      ;; vascular side membrane 
      d/dt(C_EDG[1..N_Organs,VM]) = (CL_up[i]*FR*C_EDG[i,V] ;; from vascular space
        - CL_Up[i]*FR*C_EDG[i,VM] ;; to vascular space
        - CL_UP[i]*FR*C_EDG[i,VM] ;; to endosomal pH=7.4
        + CL_up[i]*FR*C_EDG[i,E7b] ;; from endosomal pH=7.4 b
        - K_on_7_EDG*C_EDG[i,VM]*C_FcRn_VM[i]*V_VM[i] ;; rxn on 
        + K_off_7_EDG*C_EDG[i,Bound_VM]*V_VM[i] ;; rxn off
        )/V_VM[i];
      
      ;; endosomal pH=7.4
      d/dt(C_EDG[1..N_Organs,E7])= (CL_UP[i]*FR*C_EDG[i,VM] ;; from vascular membrane
        + CL_UP[i]*(1-FR)*C_EDG[i,ISM] ;; from is membrane
        - CL_UP[i]*C_EDG[i,E7] ;; to endosomal pH=6
        - K_on_7_EDG*C_EDG[i,E7]*C_FcRn_E7[i]*V_E7[i] ;; rxn on 
        + K_off_7_EDG*C_EDG[i,bound_E7]*V_E7[i] ;; rxn off
        )/V_E7[i];;
      
      ;; endosomal pH=6.0 a 
      d/dt(C_EDG[1..N_Organs,E6a]) = (CL_UP[i]*C_EDG[i,E7] ;; from E7
        - CL_UP[i]*C_EDG[i,E6a] ;; move to E7b or being routed to lysosomal for degradation
        - K_on_6_EDG*C_EDG[i,E6a]*C_FcRn_E6a[i]*V_E6a[i] ;; rxn on 
        + K_off_6_EDG*C_EDG[i,Bound_E6a]*V_E6a[i] ;; rxn off
        )/V_E6a[i]; 
      
      ;; endosomal pH=7.4  b 
      d/dt(C_EDG[1..N_Organs,E7b]) = (CL_UP[i]*(1-Prob_deg)*C_EDG[i,E6a]  ;; from 6a
        - CL_up[i]*C_EDG[i,E7b] ;; leave from E7b to membranes 
        - K_on_7_EDG*C_EDG[i,E7b]*C_FcRn_E7b[i]*V_E7b[i] ;; rxn on 
        + K_off_7_EDG*C_EDG[i,Bound_E7b]*V_E7b[i] ;; rxn off
        )/V_E7b[i];   
              
      ;; IS side mem
      d/dt(C_EDG[1..N_Organs,ISM]) = (CL_UP[i]*(1-FR)*C_EDG[i,IntS] ;; from IS space 
        - CL_UP[i]*(1-FR)*C_EDG[i,ISM] ;; to endosomal pH=7.4
        + CL_up[i]*(1-FR)*C_EDG[i,E7b] ;; from endosomal pH=7.4 b
        - CL_Up[i]*(1-FR)*C_EDG[i,ISM] ;; to interstitial space
        - K_on_7_EDG*C_EDG[i,ISM]*C_FcRn_ISM[i]*V_ISM[i] ;; rxn on 
        + K_off_7_EDG*C_EDG[i,Bound_ISM]*V_ISM[i] ;; rxn off
        )/V_ISM[i];
    
      ;; Interstitial space
      d/dt(C_EDG[1..N_Organs,IntS])=   
        ((1-sigma_V[i])*LF[i]*C_EDG[i,V] ;; going from vascular via Lymph 
        - CL_up[i]*(1-FR)*C_EDG[i,IntS] ;; pinocytosis 
        + CL_Up[i]*(1-FR)*C_EDG[i,ISM] ;; exocytosis
        -(1-Sigma_IS[i])*LF[i]*C_EDG[i,IntS];
        )/V_IntS[i] ;;
                
      ;; ==============================================================================================
      ;; Bound Endogenous IgG 
        
      ;;;;;;;;;;;;;;;; For FcRn-IgG (bounded)
      d/dt(C_EDG[1..N_Organs,bound_VM]) = 
          (K_on_7_EDG*C_EDG[i,VM]*C_FcRn_VM[i]*V_vm[i] ;; rxn on 
        - K_off_7_EDG*C_EDG[i,Bound_VM]*V_vm[i] ;; rxn off
        - K_on_7_EDG2*C_EDG[i,Bound_VM]*C_FcRn_VM[i]*V_VM[i] ;; rxn on 
        + K_off_7_EDG2*C_EDG[i,Bound2_VM]*V_VM[i] ;; rxn off 
        + CL_UP[i]*FR*C_EDG[i,Bound_E7b]  ;; from Endosomal 6b to Vascular membrane
        - CL_UP[i]*FR*C_EDG[i,Bound_VM] ;; to Endosomal 7
        - kdeg_FcRn_Ab*C_EDG[i,Bound_VM]*V_vm[i] ;; recover FcRn when IgG-FrRn is destroyed
        )/V_VM[i] ;
      
      d/dt(C_EDG[1..N_Organs,Bound2_VM]) = 
          (K_on_7_EDG2*C_EDG[i,Bound_VM]*C_FcRn_VM[i]*V_VM[i] ;; rxn on 
        - K_off_7_EDG2*C_EDG[i,Bound2_VM]*V_VM[i] ;; rxn off 
        + CL_UP[i]*FR*C_EDG[i,Bound2_E7b]  ;; from Endosomal 6b to Vascular membrane
        - CL_UP[i]*FR*C_EDG[i,Bound2_VM] ;; to Endosomal 7
        - kdeg_FcRn_Ab*C_EDG[i,Bound2_VM]*V_VM[i] ;; recover FcRn when IgG-FrRn is destroyed
        )/V_VM[i] ;

      ;; endosomal space 
      d/dt(C_EDG[1..N_Organs,bound_E7]) = 
        (+ K_on_7_EDG*C_EDG[i,E7]*C_FcRn_E7[i]*V_E7[i] ;; rxn on 
        - K_off_7_EDG*C_EDG[i,bound_E7]*V_E7[i] ;; rxn off
        - K_on_7_EDG2*C_EDG[i,bound_E7]*C_FcRn_E7[i]*V_E7[i] ;; rxn on 
        + K_off_7_EDG2*C_EDG[i,bound2_E7]*V_E7[i] ;; rxn off 
        - CL_UP[i]*C_EDG[i,Bound_E7] ;; goes to E6
        + CL_UP[i]*FR*C_EDG[i,Bound_VM] ;; from VM 
        + CL_UP[i]*(1-FR)*C_EDG[i,Bound_ISM];  ;; from ISM
        )/V_E7[i] ;;
                
      d/dt(C_EDG[1..N_Organs,bound2_E7]) = 
        (+ K_on_7_EDG2*C_EDG[i,bound_E7]*C_FcRn_E7[i]*V_E7[i] ;; rxn on 
        - K_off_7_EDG2*C_EDG[i,bound2_E7]*V_E7[i] ;; rxn off 
        - CL_UP[i]*C_EDG[i,Bound2_E7] ;; goes to E6
        + CL_UP[i]*FR*C_EDG[i,Bound2_VM] ;; from VM 
        + CL_UP[i]*(1-FR)*C_EDG[i,Bound2_ISM];  ;; from ISM
        )/V_E7[i] ;;
        
      ;; endosomal pH=6 a
      d/dt(C_EDG[1..N_Organs,bound_E6a]) = 
        (+ K_on_6_EDG*C_EDG[i,E6a]*C_FcRn_E6a[i]*V_e6a[i] ;; rxn on 
        - K_off_6_EDG*C_EDG[i,bound_E6a]*V_e6a[i] ;; rxn off
        - K_on_6_EDG2*C_EDG[i,bound_E6a]*C_FcRn_E6a[i]*V_E6a[i] ;; rxn on 
        + K_off_6_EDG2*C_EDG[i,bound2_E6a]*V_E6a[i] ;; rxn off 
        + CL_UP[i]*C_EDG[i,Bound_E7] ;; internalization, from pH7 to pH6a
        - CL_up[i]*C_EDG[i,Bound_E6a] ;; exocytosis, from E6a to E7b
        )/V_E6a[i] ;; 
  
      d/dt(C_EDG[1..N_Organs,bound2_E6a]) = 
        (+ K_on_6_EDG2*C_EDG[i,bound_E6a]*C_FcRn_E6a[i]*V_E6a[i] ;; rxn on 
        - K_off_6_EDG2*C_EDG[i,bound2_E6a]*V_E6a[i] ;; rxn off 
        + CL_UP[i]*C_EDG[i,Bound2_E7] ;; internalization, from pH7 to pH6a
        - CL_up[i]*C_EDG[i,Bound2_E6a] ;; exocytosis, from E6a to E7b
        )/V_E6a[i] ; 
  
      ;; endosomal pH=7.4  b
      d/dt(C_EDG[1..N_Organs,bound_E7b]) = 
        (+ K_on_7_EDG*C_EDG[i,E7b]*C_FcRn_E7b[i]*V_E7b[i] ;; rxn on 
        - K_off_7_EDG*C_EDG[i,bound_E7b]*V_E7b[i] ;; rxn off
        - K_on_7_EDG2*C_EDG[i, bound_E7b]*C_FcRn_E7b[i]*V_E7b[i] ;; rxn on 
        + K_off_7_EDG2*C_EDG[i,bound2_E7b]*V_E7b[i] ;; rxn off 
        + CL_UP[i]*C_EDG[i,Bound_E6a] ;; internalization, from E6a to E7b
        - CL_up[i]*C_EDG[i,Bound_E7b] ;; exocytosis, from E7b to vascular/interstitial membrane
        )/V_E7b[i] ; 
        
      d/dt(C_EDG[1..N_Organs,bound2_E7b]) = 
        (+ K_on_7_EDG2*C_EDG[i, bound_E7b]*C_FcRn_E7b[i]*V_E7b[i] ;; rxn on 
        - K_off_7_EDG2*C_EDG[i,bound2_E7b]*V_E7b[i] ;; rxn off 
        + CL_UP[i]*C_EDG[i,Bound2_E6a] ;; internalization, from E6a to E7b
        - CL_up[i]*C_EDG[i,Bound2_E7b] ;; exocytosis, from E7b to vascular/interstitial membrane
        )/V_E7b[i] ; 
        
      ;; Interstitial membrane
      d/dt(C_EDG[1..N_Organs,bound_ISM]) = 
        (+ K_on_7_EDG*C_EDG[i,ISM]*C_FcRn_ISM[i]*V_ism[i] ;; rxn on  
        - K_off_7_EDG*C_EDG[i,Bound_ISM]*V_ism[i] ;; rxn off
        - K_on_7_EDG2*C_EDG[i,Bound_ISM]*C_FcRn_ISM[i]*V_ISM[i] ;; rxn on 
        + K_off_7_EDG2*C_EDG[i,Bound2_ISM]*V_ISM[i] ;; rxn off 
        + CL_UP[i]*(1-FR)*C_EDG[i,Bound_E7b]  ;; from Endosomal 6 to interstitial membrane
        - CL_UP[i]*(1-FR)*C_EDG[i,Bound_ISM] ;; to Endosomal 7.4
        - kdeg_FcRn_Ab*C_EDG[i,Bound_ISM]*V_ism[i]  ;; recover FcRn when IgG-FrRn is destroyed
        )/V_ISM[i] ;
        
      d/dt(C_EDG[1..N_Organs,bound2_ISM]) = 
        (+ K_on_7_EDG2*C_EDG[i,Bound_ISM]*C_FcRn_ISM[i]*V_ISM[i] ;; rxn on 
        - K_off_7_EDG2*C_EDG[i,Bound2_ISM]*V_ISM[i] ;; rxn off 
        + CL_UP[i]*(1-FR)*C_EDG[i,Bound2_E7b]  ;; from Endosomal 7.4 b to interstitial membrane
        - CL_UP[i]*(1-FR)*C_EDG[i,Bound2_ISM] ;; to Endosomal 7.4
        - kdeg_FcRn_Ab*C_EDG[i,Bound2_ISM]*V_ism[i]  ;; recover FcRn when IgG-FrRn is destroyed
        )/V_ISM[i] ;
        
      ;; ==============================================================================================
      ;; Free FcRn in Organs
  
      ;; vascular side membrane 
      d/dt(C_FcRn_VM[1..N_Organs]) = 
        (+ CL_up[i]*FR*C_FcRn_E7b[i]*(1-FcRn_recycle_fraction); from endosomal pH=7.4  b
        - CL_Up[i]*FR*C_FcRn_VM[i] ;; to vascular space
        ;; EDG 
        +v_vm[i]*(
        - K_on_7_EDG*C_EDG[i,VM]*C_FcRn_VM[i] ;; rxn on in vascular side membrane 
        + K_off_7_EDG*C_EDG[i,Bound_VM] ;; rxn off vascular side membrane 
        + kdeg_FcRn_Ab*C_EDG[i,Bound_VM]
        - K_on_7_EDG2*C_EDG[i,Bound_VM]*C_FcRn_VM[i] ;; rxn on in vascular side membrane 
        + K_off_7_EDG2*C_EDG[i,Bound2_VM] ;; rxn off vascular side membrane 
        + 2*kdeg_FcRn_Ab*C_EDG[i,Bound2_VM] ;; 
        ;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;
        ;; EXG 
        - K_on_7_EXG*C_EXG[1,i,VM]*C_FcRn_VM[i] ;; rxn on in vascular side membrane 
        + K_off_7_EXG*C_EXG[1,i,Bound_VM] ;; rxn off vascular side membrane 
        + kdeg_FcRn_Ab*C_EXG[1,i,Bound_VM]
        - K_on_7_EXG2*C_EXG[1,i,Bound_VM]*C_FcRn_VM[i] ;; rxn on in vascular side membrane 
        + K_off_7_EXG2*C_EXG[1,i,Bound2_VM] ;; rxn off vascular side membrane 
        + 2*kdeg_FcRn_Ab*C_EXG[1,i,Bound2_VM] ;; 
        ))/V_VM[i];
      
      ;; endosomal pH=7.4
      d/dt(C_FcRn_E7[1..N_Organs]) = 
                 (CL_UP[i]*FR*C_FcRn_VM[i] ;; from vascular membrane
        + CL_UP[i]*(1-FR)*C_FcRn_ISM[i] ;; from is membrane
        - CL_UP[i]*C_FcRn_E7[i] ;; to endosomal pH=6
        + V_E7[i]*(
        - K_on_7_EDG*C_EDG[i,E7]*C_FcRn_E7[i] ;; rxn on endosomal pH=7.4
        + K_off_7_EDG*C_EDG[i,bound_E7] ;; rxn off endosomal pH=7.4.
        - K_on_7_EDG2*C_EDG[i,bound_E7]*C_FcRn_E7[i] ;; rxn on endosomal pH=7.4. 
        + K_off_7_EDG2*C_EDG[i,bound2_E7] ;; rxn off endosomal pH=7.4. 
        ;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;
        ;; EXG
        - K_on_7_EXG*C_EXG[1,i,E7]*C_FcRn_E7[i] ;; rxn on endosomal pH=7.4
        + K_off_7_EXG*C_EXG[1,i,bound_E7] ;; rxn off endosomal pH=7.4
        - K_on_7_EXG2*C_EXG[1,i,bound_E7]*C_FcRn_E7[i] ;; rxn on endosomal pH=7.4. 
        + K_off_7_EXG2*C_EXG[1,i,bound2_E7] ;; rxn off endosomal pH=7.4. 
        ))/V_E7[i];;
      
      ;; endosomal pH=6.0

        - CL_UP[i]*C_FcRn_E6a[i] ;; from E6a to E7b
        + CL_up[i]*C_FcRn_E7b[i]*FcRn_recycle_fraction
        + V_E6a[i]*(
        - K_on_6_EDG*C_EDG[i,E6a]*C_FcRn_E6a[i] ;; rxn on endosomal pH=6.0
        + K_off_6_EDG*C_EDG[i,Bound_E6a] ;; rxn off endosomal pH=6.0
        - K_on_6_EDG2*C_EDG[i,Bound_E6a]*C_FcRn_E6a[i] ;; rxn on endosomal pH=6.0. 
        + K_off_6_EDG2*C_EDG[i,Bound2_E6a] ;; rxn off endosomal pH=6.0. 
        ;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;
        ;; EXG
        - K_on_6_EXG*C_EXG[1,i,E6a]*C_FcRn_E6a[i] ;; rxn on endosomal pH=6.0
        + K_off_6_EXG*C_EXG[1,i,Bound_E6a] ;; rxn off endosomal pH=6.0
        - K_on_6_EXG2*C_EXG[1,i,Bound_E6a]*C_FcRn_E6a[i] ;; rxn on endosomal pH=6.0. 
        + K_off_6_EXG2*C_EXG[1,i,Bound2_E6a] ;; rxn off endosomal pH=6.0. 
        ))/V_E6a[i] ; 
    
      ;; endosomal pH=7.4
      d/dt(C_FcRn_E7b[1..N_Organs]) = (CL_UP[i]*C_FcRn_E6a[i] ;; from E6a
        - CL_UP[i]*C_FcRn_E7b[i] ;; from E7b to membranes
        + V_E7b[i]*(
        - K_on_7_EDG*C_EDG[i,E7b]*C_FcRn_E7b[i] ;; rxn on endosomal pH=7.4
        + K_off_7_EDG*C_EDG[i,Bound_E7b] ;; rxn off endosomal pH=7.4
        - K_on_7_EDG2*C_EDG[i,Bound_E7b]*C_FcRn_E7b[i] ;; rxn on endosomal pH=7.4. 
        + K_off_7_EDG2*C_EDG[i,Bound2_E7b] ;; rxn off endosomal pH=7.4. 
        ;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;
        ;; EXG 
        - K_on_7_EXG*C_EXG[1,i,E7b]*C_FcRn_E7b[i] ;; rxn on endosomal pH=7.4
        + K_off_7_EXG*C_EXG[1,i,Bound_E7b] ;; rxn off endosomal pH=7.4
        - K_on_7_EXG2*C_EXG[1,i,Bound_E7b]*C_FcRn_E7b[i] ;; rxn on endosomal pH=7.4. 
        + K_off_7_EXG2*C_EXG[1,i,Bound2_E7b] ;; rxn off endosomal pH=7.4. 
        ))/V_E7b[i]; 
        
      ;; IS side mem
      d/dt(C_FcRn_ISM[1..N_Organs]) = 
        (- CL_UP[i]*(1-FR)*C_FcRn_ISM[i] ;; from IS membrane to endosomal pH=7.4
        + CL_up[i]*(1-FR)*C_FcRn_E7b[i]*(1-FcRn_recycle_fraction) ;; from endosomal pH=7.4  b to IS side mem
        + V_ISM[i]*(
        - K_on_7_EDG*C_EDG[i,ISM]*C_FcRn_ISM[i] ;; rxn on IS side mem
        + K_off_7_EDG*C_EDG[i,Bound_ISM] ;; rxn off IS side mem
        + kdeg_FcRn_Ab*C_EDG[i,Bound_ISM]
        - K_on_7_EDG2*C_EDG[i,Bound_ISM]*C_FcRn_ISM[i] ;; rxn on IS side mem. 
        + K_off_7_EDG2*C_EDG[i,Bound2_ISM] ;; rxn off IS side mem. 
        + 2*kdeg_FcRn_Ab*C_EDG[i,Bound2_ISM] ;; 
        ;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;
        ;; EXG 
        - K_on_7_EXG*C_EXG[1,i,ISM]*C_FcRn_ISM[i] ;; rxn on IS side mem
        + K_off_7_EXG*C_EXG[1,i,Bound_ISM] ;; rxn off IS side mem
        + kdeg_FcRn_Ab*C_EXG[1,i,Bound_ISM]
        - K_on_7_EXG2*C_EXG[1,i,Bound_ISM]*C_FcRn_ISM[i] ;; rxn on IS side mem. 
        + K_off_7_EXG2*C_EXG[1,i,Bound2_ISM] ;; rxn off IS side mem. 
        + 2*kdeg_FcRn_Ab*C_EXG[1,i,Bound2_ISM] ;; 
        ))/V_ISM[i];
      
 
;; ==============================================================================================  
;; Main Plasma Compartment    
                  
      ;;; flux of endogenous IgG in Lymph node
      d/dt(C_EDG_LN) = 
          ((1-Sigma_IS[Heart])*LF[Heart]*C_EDG[Heart, IntS]
        + (1-Sigma_IS[Kidney])*LF[Kidney]*C_EDG[Kidney, IntS]
        + (1-Sigma_IS[Muscle])*LF[Muscle]*C_EDG[Muscle, IntS]
        + (1-Sigma_IS[Skin])*LF[Skin]*C_EDG[Skin, IntS]
        + (1-Sigma_IS[Brain])*LF[Brain]*C_EDG[Brain, IntS]
        + (1-Sigma_IS[Adipose])*LF[Adipose]*C_EDG[Adipose, IntS]
        + (1-Sigma_IS[Thymus])*LF[Thymus]*C_EDG[Thymus, IntS]
        + (1-Sigma_IS[Liver])*LF[Liver]*C_EDG[Liver, IntS]
        + (1-Sigma_IS[Spleen])*LF[Spleen]*C_EDG[Spleen, IntS]
        + (1-Sigma_IS[Pancreas])*LF[Pancreas]*C_EDG[Pancreas, IntS]
        + (1-Sigma_IS[SI])*LF[SI]*C_EDG[SI, IntS]
        + (1-Sigma_IS[LI])*LF[LI]*C_EDG[LI, IntS]
        + (1-Sigma_IS[Bone])*LF[Bone]*C_EDG[Bone, IntS]
        + (1-Sigma_IS[Other])*LF[Other]*C_EDG[Other, IntS]
        + (1-Sigma_IS[Lung])*LF[Lung]*C_EDG[Lung, IntS]
        - L_LymphNode*C_EDG_LN)/V_LN;         
            
      ;;; flux of exogenous IgG in Plasma
      w_other_1_EXG_Plasma = 
        (PLQ[Heart]-LF[Heart])*C_EXG[1,Heart,V] 
        +(PLQ[Kidney]-LF[Kidney])*C_EXG[1,Kidney,V]
        +(PLQ[Muscle]-LF[Muscle])*C_EXG[1,Muscle,V]

           +(PLQ[Spleen]-LF[Spleen])*C_EXG[1,Liver,V]
        +(PLQ[Pancreas]-LF[Pancreas])*C_EXG[1,Liver,V]
        +(PLQ[SI]-LF[SI])*C_EXG[1,Liver,V]
        +(PLQ[LI]-LF[LI])*C_EXG[1,Liver,V]
        +(PLQ[Bone]-LF[Bone])*C_EXG[1,Bone,V]
        +(PLQ[Other]-LF[Other])*C_EXG[1,Other,V]
        -(PLQ[Lung]+LF[Lung])*C_EXG_Plasma
        +L_LymphNode*C_EXG_LN 

		w_other_2_EXG_Plasma = 0 

		w_other_3_EXG_Plasma = 0
        

      ;;; flux of exogenous IgG in Lymph node
      d/dt(C_EXG_LN) = 
          ((1-Sigma_IS[Heart])*LF[Heart]*C_EXG[1,Heart, IntS]
        + (1-Sigma_IS[Kidney])*LF[Kidney]*C_EXG[1,Kidney, IntS]
        + (1-Sigma_IS[Muscle])*LF[Muscle]*C_EXG[1,Muscle, IntS]
        + (1-Sigma_IS[Skin])*LF[Skin]*C_EXG[1,Skin, IntS]
        + (1-Sigma_IS[Brain])*LF[Brain]*C_EXG[1,Brain, IntS]
        + (1-Sigma_IS[Adipose])*LF[Adipose]*C_EXG[1,Adipose, IntS]
        + (1-Sigma_IS[Thymus])*LF[Thymus]*C_EXG[1,Thymus, IntS]
        + (1-Sigma_IS[Liver])*LF[Liver]*C_EXG[1,Liver, IntS]
        + (1-Sigma_IS[Spleen])*LF[Spleen]*C_EXG[1,Spleen, IntS]
        + (1-Sigma_IS[Pancreas])*LF[Pancreas]*C_EXG[1,Pancreas, IntS]
        + (1-Sigma_IS[SI])*LF[SI]*C_EXG[1,SI, IntS]
        + (1-Sigma_IS[LI])*LF[LI]*C_EXG[1,LI, IntS]
        + (1-Sigma_IS[Bone])*LF[Bone]*C_EXG[1,Bone, IntS]
        + (1-Sigma_IS[Other])*LF[Other]*C_EXG[1,Other, IntS]
        + (1-Sigma_IS[Lung])*LF[Lung]*C_EXG[1,LUng, IntS]
        - L_LymphNode*C_EXG_LN
                )/V_LN ; 
      
	d/dt(C_EDG_Plasma) = 0/V_plasma

    d/dt(AUC_free_mAb) = C_EXG_Plasma_uM 

	
	;; SIMULATION EQUATIONS
    
    PS_Kd = 10^(EXP(PS_a - PS_b*PS_Score)) 
    
    Organ_Endothelial_Cell[1..N_Organs]  = Total_Endothelial_Cell*Endothelial_Cell_Frac[i]*Scale_Factor;;

    V_ISM [1..N_Organs]        = CL_UP[i]*tau_ISM

    ;;; Readout variables ;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;
	
    C_plasma_readout = C_EXG_Plasma*1E3;;
    EXG_ugml_Readout = C_EXG_Plasma*MW_EDG/1000;;  ;;
    FcRn_in_uM = FcRn_Conc ;;
            
            ;;  DUMMY VARIABLES FOR SUMMATION
            CV_VM[1..N_Organs] = C_FcRn_VM[i]*V_VM[i] ;;

            ;;
			
      Free_FcRn[FcRn_VM] = ARRAYSUM(CV_VM[*])*1E3;;

      Free_FcRn_Pct[1..5] = Free_FcRn[i]/Total_FcRn[i];

        
        ;;  DUMMY VARIABLES FOR SUMMATION
        C_EDG_bVM[1..N_Organs,Bound_VM]       = C_EDG[i,Bound_VM]

        C_EXG_1_bE7[1..N_Organs]            = C_EXG[1,i,Bound_E7]

				+ V_VM[i]*C_EXG[1,i,Bound_VM] + V_E7[i]*C_EXG[1,i,Bound_E7] + V_E6a[i]*C_EXG[1,i,Bound_E6a]
				+ V_E7b[i]*C_EXG[1,i,Bound_E7b] + V_ISM[i]*C_EXG[1,i,Bound_ISM] 
				+ V_VM[i]*C_EXG[1,i,Bound2_VM] + V_E7[i]*C_EXG[1,i,Bound2_E7] + V_E6a[i]*C_EXG[1,i,Bound2_E6a]
				+ V_E7b[i]*C_EXG[1,i,Bound2_E7b] + V_ISM[i]*C_EXG[1,i,Bound2_ISM] 
				+ V_VM[i]*C_EXG[1,i,Bound_VM_mem] + V_ISM[i]*C_EXG[1,i,Bound_ISM_mem] ;;;

		mAb_Biodistr_Coeff_in_tissue[1..N_Organs] = (Total_EXG_intra_tissue[i]/V_Organ[i])/(1e-9 + C_EXG_Plasma) ; 

      Total_FcRn[FcRn_VM] = ((ARRAYSUM(C_FcRn_VM[*])

      Total_FcRn_MB = ARRAYSUM(Total_FcRn[*])
